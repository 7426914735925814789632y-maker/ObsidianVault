// 1. 集約したいフォルダのパスを正確に指定（例: "意味へいシリーズ" など）
const folderPath = "山口晃央/FOS［山口晃央］/(ⅰ)_🌟-④/│アイデア/│動画企画/│オリジナルゲーム/│意味へいゲーム"; 

// 2. エラーを回避するため、現在のノートのパスを安全に取得
const currentFile = dv.current() ? dv.current().file?.path : null;

// 3. 指定したフォルダ内のページを取得
const pages = dv.pages(`"${folderPath}"`)
    .filter(p => p.file && p.file.path !== currentFile) // 自身のノートは除外
    .sort(p => p.file.name, 'asc'); // 名前順にソート

// 4. まとめて埋め込み表示
if (pages.length === 0) {
    dv.paragraph("⚠️ 指定されたフォルダに有効なノートが見つかりませんでした。フォルダ名を確認してください。");
} else {
    pages.forEach(p => {
        dv.paragraph(`![[${p.file.name}]]`);
    });
}
