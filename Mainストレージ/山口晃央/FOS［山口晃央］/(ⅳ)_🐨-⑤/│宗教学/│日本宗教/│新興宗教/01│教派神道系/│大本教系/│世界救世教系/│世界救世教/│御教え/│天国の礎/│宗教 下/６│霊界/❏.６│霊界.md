// 1. 集約したいフォルダのパスを正確に指定
const folderPath = "山口晃央/FOS［山口晃央］/(ⅳ)_🐨-⑤/│宗教学/│日本宗教/│新興宗教/01│教派神道系/│大本教系/│世界救世教系/│世界救世教/│御教え/│天国の礎/│宗教 下/６│霊界";

// 2. エラーを回避するため、現在のノートのパスを安全に取得
const currentFile = dv.current() ? dv.current().file?.path : null;

// 3. 指定したフォルダ内のページを取得（直下の階層のみ）
const pages = dv.pages(`"${folderPath}"`)
    .filter(p => {
        // ファイルが存在し、かつ自分自身ではないこと
        if (!p.file || p.file.path === currentFile) return false;
        
        // 【修正箇所】ファイルの親フォルダ（p.file.folder）が、指定したフォルダと完全に一致するものだけを許可
        return p.file.folder === folderPath;
    })
    .sort(p => p.file.name, 'asc'); // 名前順にソート

// 4. まとめて埋め込み表示
if (pages.length === 0) {
    dv.paragraph("⚠️ 指定されたフォルダの直下に有効なノートが見つかりませんでした。フォルダ名を確認してください。");
} else {
    pages.forEach(p => {
        // 安全にリンクを埋め込むため、ファイル名ではなくパス、または特別な識別子での指定が確実です
        dv.paragraph(`![[${p.file.name}]]`);
    });
}
