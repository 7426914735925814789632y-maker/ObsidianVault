// 1. 集約したいフォルダのパスを正確に指定
const folderPath = "山口晃央/Past：山口晃央の過去/DFB/〔2026年〕/〔6月〕";

// 2. エラーを回避するため、現在のノートのパスを安全に取得
const currentFile = dv.current() ? dv.current().file?.path : null;

// 3. 指定したフォルダ内のページを取得（フルパス順にソートすることで、自動的に日付フォルダ順になります）
const pages = dv.pages(`"${folderPath}"`)
    .filter(p => p.file && p.file.path !== currentFile) // 自身のノートは除外
    .sort(p => p.file.path, 'asc'); // ファイル名ではなく「フォルダのパス（日付）」順にソート

// 4. まとめて埋め込み表示
if (pages.length === 0) {
    dv.paragraph("⚠️ 指定されたフォルダに有効なノートが見つかりませんでした。フォルダ名を確認してください。");
} else {
    pages.forEach(p => {
        // 深い階層でも100%特定できるよう、拡張子(.md)を除いた正確なフルパスに変換
        const targetPath = p.file.path.replace(/\.md$/, "");
        dv.paragraph(`![[${targetPath}]]`);
    });
}
