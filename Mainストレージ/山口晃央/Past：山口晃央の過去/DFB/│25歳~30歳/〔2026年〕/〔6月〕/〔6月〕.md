```dataview
TABLE WITHOUT ID folderLink AS "フォルダ", choice(length(noteLinks) > 0, noteLinks, "ー") AS "ノート"
WHERE (this.file.folder = "" OR file.folder = this.file.folder OR startswith(file.folder, this.file.folder + "/"))
  AND choice(file.folder = "", 0, length(split(file.folder, "/"))) <= choice(this.file.folder = "", 0, length(split(this.file.folder, "/"))) + 2
  AND file.path != this.file.path
GROUP BY file.folder
FLATTEN choice(key = "", "ルート", regexreplace(key, ".*/", "")) AS folderName
FLATTEN [ filter(rows.file, (x) => x.name = folderName)[0] ] AS indexFile
FLATTEN choice(indexFile, link(indexFile.path, folderName), folderName) AS folderLink
FLATTEN [ filter(rows.file, (x) => x.name != folderName).link ] AS noteLinks
SORT key ASC
```
