// 1. 集約したいフォルダのパスを指定（対象ファイルの同階層フォルダ）
const folderPath = "山口晃央/ToolBox/│AI_マルチプロンプト/│Gemini用プロンプト/│非エンジニア用・システム開発指南書";

// 2. エラーを回避するため、現在のノートのパスを安全に取得
const currentFile = dv.current() ? dv.current().file?.path : null;

// 3. 指定したフォルダ直下のページを取得（自身は除外、名前順ソート）
const pages = dv.pages(`"${folderPath}"`)
    .filter(p => p.file && p.file.folder === folderPath && p.file.path !== currentFile)
    .sort(p => p.file.name, 'asc');

if (pages.length === 0) {
    dv.paragraph("⚠️ 指定されたフォルダに有効なノートが見つかりませんでした。フォルダ名を確認してください。");
} else {
    // --- UI：一括コピー用コントロールエリア ---
    const uiBox = dv.container.createDiv({
        attr: { style: "display: flex; gap: 8px; flex-wrap: wrap; align-items: center; margin-bottom: 12px; padding: 10px; border-radius: 8px; background: var(--background-secondary);" }
    });

    const copyBtn = uiBox.createEl("button", {
        text: `📋 全文を一括コピー (${pages.length}件)`,
        attr: { style: "cursor: pointer; background: var(--interactive-accent); color: var(--text-on-accent); border: none; padding: 6px 12px; border-radius: 6px; font-weight: 600;" }
    });

    const toggleBtn = uiBox.createEl("button", {
        text: "📝 コピー用テキストを表示",
        attr: { style: "cursor: pointer; padding: 6px 12px; border-radius: 6px;" }
    });

    const previewArea = dv.container.createEl("textarea", {
        attr: {
            readonly: "true",
            style: "display: none; width: 100%; height: 320px; margin-bottom: 16px; padding: 10px; font-family: var(--font-monospace); font-size: 0.85em; border-radius: 6px; box-sizing: border-box;"
        }
    });

    // 全ノートの本文を読み込んで連結する関数
    const buildCombinedText = async () => {
        const parts = [];
        for (const p of pages) {
            const tfile = app.vault.getAbstractFileByPath(p.file.path);
            if (tfile) {
                const raw = await app.vault.read(tfile);
                parts.push(`# ${p.file.name}\n\n${raw.trim()}`);
            }
        }
        return parts.join("\n\n---\n\n");
    };

    // ワンクリックコピー動作
    copyBtn.addEventListener("click", async () => {
        const originalText = copyBtn.textContent;
        copyBtn.textContent = "⏳ 読み込み中...";
        try {
            const text = await buildCombinedText();
            try {
                await navigator.clipboard.writeText(text);
            } catch (e) {
                previewArea.style.display = "block";
                previewArea.value = text;
                previewArea.select();
                document.execCommand("copy");
            }
            copyBtn.textContent = "✅ コピー完了！";
            if (typeof Notice !== "undefined") new Notice(`✅ ${pages.length}件のノートを一括コピーしました`);
        } catch (err) {
            copyBtn.textContent = "❌ エラー";
            console.error(err);
        }
        setTimeout(() => { copyBtn.textContent = originalText; }, 2000);
    });

    // テキストエリア表示切り替え（部分コピー・目視確認用）
    toggleBtn.addEventListener("click", async () => {
        if (previewArea.style.display === "none") {
            previewArea.value = await buildCombinedText();
            previewArea.style.display = "block";
            toggleBtn.textContent = "🔽 コピー用テキストを隠す";
        } else {
            previewArea.style.display = "none";
            toggleBtn.textContent = "📝 コピー用テキストを表示";
        }
    });

    // 4. まとめて埋め込み表示
    pages.forEach(p => {
        dv.paragraph(`![[${p.file.name}]]`);
    });
}
