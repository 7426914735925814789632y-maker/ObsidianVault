
<%*
// 現在アクティブなエディタ（ノート）を取得
const markdownView = app.workspace.getActiveViewOfType(tp.obsidian.MarkdownView);

if (markdownView) {
    const editor = markdownView.editor;
    const currentText = editor.getValue();
    
    // \x5B = '［', \x5D = '］' の文字コードを指定して置換
    const fixedText = currentText
        .replace(/\x5B/g, '［')
        .replace(/\x5D/g, '］');
    
    // エディタの内容を書き換え（Ctrl+Z で元に戻せます）
    editor.setValue(fixedText);
    new Notice("すべての半角括弧を全角に変換しました。");
} else {
    new Notice("アクティブなノートが見つかりません。");
}
-%>
