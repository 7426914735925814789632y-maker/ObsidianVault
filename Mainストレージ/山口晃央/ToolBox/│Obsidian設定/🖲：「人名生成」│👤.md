<%*
while (true) {
    // 1. 保管庫内のすべてのフォルダを取得してリストアップする
    const files = app.vault.getAllLoadedFiles();
    const folders = files.filter(f => f.children && f.path !== "/").map(f => f.path);
    folders.unshift("/"); // ルートフォルダを選択肢に追加

    // 2. ユーザーに作成先のフォルダを検索・選択してもらう（Esc等で完全にキャンセルされたら終了）
    const selectedFolder = await tp.system.suggester(folders, folders, false, "【連続入力中】作成先のフォルダを選択（Escで終了 / 姓名空欄で終了）");
    if (selectedFolder === undefined || selectedFolder === null) break;

    // 3. ユーザーに姓と名を別々に入力してもらう
    const lastName = (await tp.system.prompt("【姓（苗字）】を入力してください（終了する場合は空欄のままエンター）:", "") || "").trim();
    const firstName = (await tp.system.prompt("【名前】を入力してください（終了する場合は空欄のままエンター）:", "") || "").trim();

    // ★ガード：姓名のどちらも入力されなかった（エンターを連打した）場合は、ここで安全に処理を停止
    if (!lastName && !firstName) {
        new Notice("入力を検知しなかったため、連続作成処理を終了しました。");
        break;
    }

    // 姓名を結合するロジック
    let name = "";
    if (lastName && firstName) {
        name = `${lastName} ${firstName}`;
    } else {
        name = lastName || firstName;
    }

    // 4. 生成するフォルダ・ファイル名の定義
    const folderName = `👤：「${name}」`;
    const waypointFileName = `👤：「${name}」.md`; 
    const baseFileName = `☷│👤：「${name}」.base`;

    // 5. 選択されたフォルダパスを基に、新規フォルダのフルパスを構築
    const prefixPath = selectedFolder === "/" ? "" : selectedFolder + "/";
    const targetFolderPath = `${prefixPath}${folderName}`;

    // 6. 新規人名フォルダの作成
    await app.vault.createFolder(targetFolderPath);

    /* ==========================================================
       ファイル 1: 人名ファイル（＝ %% Waypoint %% ファイル。空行フォーマット死守）
       ========================================================== */
    const waypointPath = `${targetFolderPath}/${waypointFileName}`;
    const waypointTag = "%" + "% Waypoint %" + "%";
    const waypointContent = `---
意味:

生年月日:

出身:

大学:

---
${waypointTag}
`;
    const createdWaypointFile = await app.vault.create(waypointPath, waypointContent);

    /* ==========================================================
       ファイル 2: .base ファイルの作成（YAMLフィルター構造を完全死守）
       ========================================================== */
    const targetMainFilePath = `${targetFolderPath}/👤：「${name}」`;
    const basePath = `${targetFolderPath}/${baseFileName}`;

    const baseContent = `views:
  - type: table
    name: 表
    filters:
      and:
        - file.hasLink("${targetMainFilePath}")
`;
    await app.vault.create(basePath, baseContent);

    /* ==========================================================
       作成した人名ファイルを「新規タブ」でバックグラウンドオープン
       ========================================================== */
    if (createdWaypointFile) {
        await app.workspace.getLeaf('tab').openFile(createdWaypointFile);
    }
}
