<%*
// 1. 選択範囲があればそれを取得、なければクリップボードから取得
let text = tp.file.selection();
if (!text) {
    text = await tp.system.clipboard();
}

if (text) {
    let formatted = text
        // 全角・半角スペースの連続を1つの改行に置換
        .replace(/[  ]+/g, '\n')
        // 3つ以上連続してしまった改行を、美しい2連改行（空行1行分）に圧縮
        .replace(/\n{3,}/g, '\n\n')
        // 行頭や行末に発生した不要な空白をトリミング
        .trim();

    // 処理したテキストを現在のカーソル位置に挿入
    tR += formatted;
} else {
    new Notice("テキストが選択されていない、またはクリップボードが空です。");
}
-%>
