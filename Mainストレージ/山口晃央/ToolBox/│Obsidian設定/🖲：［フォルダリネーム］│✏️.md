// 共通のリネーム処理関数
const processFolderRename = async (folder) => {
    const oldName = folder.name;
    const newName = await tp.system.prompt(`「${oldName}」の新しいフォルダ名 (Enterのみでスキップ):`, oldName);

    // 空欄、または変更がない場合はスキップ
    if (!newName || newName === oldName) {
        new Notice(`「${oldName}」はスキップしました。`);
        return false;
    }

    const itemsToRename = [];
    const collectItems = (f) => {
        for (let child of f.children) {
            if (child.name.includes(oldName)) {
                itemsToRename.push(child);
            }
            if (child.children) {
                collectItems(child);
            }
        }
    };

    collectItems(folder);
    itemsToRename.sort((a, b) => b.path.length - a.path.length);

    for (let item of itemsToRename) {
        const newItemName = item.name.split(oldName).join(newName);
        const parentPath = item.parent.path === "/" ? "" : item.parent.path + "/";
        await app.fileManager.renameFile(item, parentPath + newItemName);
    }

    const folderParentPath = folder.parent.path === "/" ? "" : folder.parent.path + "/";
    await app.fileManager.renameFile(folder, folderParentPath + newName);
    
    new Notice(`「${oldName}」→「${newName}」のリネームが完了しました。`);
    return true;
};

// 1. Vault内の全フォルダを取得
const allFolders = app.vault.getAllLoadedFiles().filter(f => f.children && f.path !== "/");
const folderPaths = allFolders.map(f => f.path);

// 2. 最初のフォルダを選択
const targetFolder = await tp.system.suggester(folderPaths, allFolders, false, "最初にリネームするフォルダを検索・選択");
if (!targetFolder) {
    new Notice("フォルダ選択がキャンセルされました。");
    return;
}

// 3. 最初のフォルダを処理
await processFolderRename(targetFolder);

// 4. 同じ階層の他のフォルダがあるか確認
const parentFolder = targetFolder.parent;
const siblings = parentFolder.children.filter(f => f.children && f.path !== targetFolder.path);

if (siblings.length > 0) {
    const shouldContinue = await tp.system.suggester(
        ["はい（同じ階層の残りのフォルダを順次処理）", "いいえ（ここで終了）"], 
        [true, false], 
        false, 
        `「${parentFolder.name || "ルート"}」内の他のフォルダも続けて確認・リネームしますか？`
    );

    if (shouldContinue) {
        for (let sibling of siblings) {
            await processFolderRename(sibling);
        }
        new Notice("同じ階層のフォルダの連続リネーム処理が終了しました。");
    } else {
        new Notice("処理を終了しました。");
    }
} else {
    new Notice("同じ階層に他のフォルダはありません。処理を終了します。");
}
