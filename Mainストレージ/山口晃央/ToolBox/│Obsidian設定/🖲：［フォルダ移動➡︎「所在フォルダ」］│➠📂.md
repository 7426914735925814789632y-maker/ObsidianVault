<%*
const IGNORE_DIRS = [".obsidian", ".trash", "ToolBox", ".github"];
const WP_TAG = "\u0025\u0025 Waypoint \u0025\u0025";

function isIgnored(path) {
    const parts = path.split("/");
    for (let i = 0; i < parts.length; i++) {
        if (IGNORE_DIRS.indexOf(parts[i]) !== -1) {
            return true;
        }
    }
    return false;
}

function findDestFolder(targetName, allFolders) {
    let candidates = allFolders.filter(function(f) {
        return f.name === targetName;
    });

    const keywords = targetName.split(/[\s　]+/).filter(function(k) {
        return k.length > 0;
    });

    if (candidates.length === 0) {
        candidates = allFolders.filter(function(f) {
            return keywords.every(function(kw) {
                return f.name.indexOf(kw) !== -1;
            });
        });
    }

    if (candidates.length === 0) {
        candidates = allFolders.filter(function(f) {
            return keywords.every(function(kw) {
                return f.path.indexOf(kw) !== -1;
            });
        });
    }

    candidates.sort(function(a, b) {
        return a.path.split("/").length - b.path.split("/").length;
    });

    return candidates.length > 0 ? candidates[0] : null;
}

let processedCount = 0;
let createdFolderCount = 0;

const allLoaded = app.vault.getAllLoadedFiles();
const allFolders = [];
for (let i = 0; i < allLoaded.length; i++) {
    const item = allLoaded[i];
    if (item.children !== undefined && !isIgnored(item.path)) {
        allFolders.push(item);
    }
}

const allFiles = app.vault.getMarkdownFiles();

for (let i = 0; i < allFiles.length; i++) {
    const file = allFiles[i];
    if (isIgnored(file.path)) continue;

    const cache = app.metadataCache.getFileCache(file);
    const fm = cache && cache.frontmatter ? cache.frontmatter : null;
    if (!fm || !fm["所在フォルダ"]) continue;

    const targetName = String(fm["所在フォルダ"]).trim();
    // 空欄、「【完了済み】」、「【不定】」は処理対象外としてスキップ
    if (!targetName || targetName === "【完了済み】" || targetName === "【不定】") continue;

    const isWpTrue = (fm["Waypoint"] === true || fm["Waypoint"] === "true");
    const destFolder = findDestFolder(targetName, allFolders);

    if (destFolder) {
        if (isWpTrue && file.parent.name === file.basename) {
            const srcFolder = file.parent;
            const finalFolderPath = destFolder.path === "/" 
                ? srcFolder.name 
                : destFolder.path + "/" + srcFolder.name;

            if (!app.vault.getAbstractFileByPath(finalFolderPath) && srcFolder.path !== destFolder.path && destFolder.path.indexOf(srcFolder.path + "/") !== 0) {
                await app.fileManager.renameFile(srcFolder, finalFolderPath);
            }
        } else {
            const finalFilePath = destFolder.path === "/" 
                ? file.name 
                : destFolder.path + "/" + file.name;

            if (!app.vault.getAbstractFileByPath(finalFilePath) && file.parent.path !== destFolder.path) {
                await app.fileManager.renameFile(file, finalFilePath);
            }
        }

        await app.fileManager.processFrontMatter(file, function(fmUpdate) {
            fmUpdate["所在フォルダ"] = "【完了済み】";
            fmUpdate["所在チェック"] = false;
        });
        processedCount++;

    } else {
        let basePath = "/";
        if (isWpTrue && file.parent.name === file.basename) {
            basePath = file.parent.parent ? file.parent.parent.path : "/";
        } else {
            basePath = file.parent ? file.parent.path : "/";
        }

        const newFolderPath = (basePath === "/" || basePath === "") 
            ? targetName 
            : basePath + "/" + targetName;

        try {
            if (!app.vault.getAbstractFileByPath(newFolderPath)) {
                await app.vault.createFolder(newFolderPath);
            }
            const newNotePath = newFolderPath + "/" + targetName + ".md";
            if (!app.vault.getAbstractFileByPath(newNotePath)) {
                const noteContent = "---\n意味: \nWaypoint: true\n---\n\n" + WP_TAG + "\n";
                await app.vault.create(newNotePath, noteContent);
            }
            createdFolderCount++;
        } catch (e) {
            new Notice("フォルダ作成エラー: " + e.message);
        }

        await app.fileManager.processFrontMatter(file, function(fmUpdate) {
            fmUpdate["所在フォルダ"] = "【完了済み】";
            fmUpdate["所在チェック"] = false;
        });
        processedCount++;
    }
}

new Notice("所在処理完了：" + processedCount + "件（新規フォルダ作成 " + createdFolderCount + "件）");
%>
