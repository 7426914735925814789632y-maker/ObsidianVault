// フォルダ取得関数（ルートを除く）
const getTargetFolders = () => {
    return app.vault.getAllLoadedFiles().filter(f => f.children && f.path !== "/");
};

let foldersToMove = [];
let keepPicking = true;

// 1. フォルダ選択ループ
while (keepPicking) {
    const folders = getTargetFolders();
    const folderPaths = folders.map(f => f.path);
    
    const selectedFolder = await tp.system.suggester(folderPaths, folders, false, "移動したいフォルダを選択してください");
    
    if (!selectedFolder) break;

    // 重複チェック
    if (!foldersToMove.some(f => f.path === selectedFolder.path)) {
        foldersToMove.push(selectedFolder);
    }

    // 同階層のフォルダをまとめて選択するか確認
    const askSiblings = await tp.system.suggester(
        [`「${selectedFolder.name}」と同階層のフォルダを全て追加`, "個別選択のみで次へ"],
        [true, false],
        false,
        "同階層のフォルダをまとめて選択しますか？"
    );

    if (askSiblings) {
        const parent = selectedFolder.parent;
        const siblings = parent.children.filter(f => f.children && f.path !== selectedFolder.path);
        
        for (let sib of siblings) {
            if (!foldersToMove.some(f => f.path === sib.path)) {
                foldersToMove.push(sib);
            }
        }
        new Notice(`${siblings.length}個の同階層フォルダを追加しました。`);
    }

    // まだ追加するか確認
    const askMore = await tp.system.suggester(
        ["はい（別のフォルダを選択）", "いいえ（移動先を選択へ）"],
        [true, false],
        false,
        `現在 ${foldersToMove.length} 個のフォルダを選択済みです。続けますか？`
    );
    keepPicking = askMore;
}

if (foldersToMove.length === 0) {
    new Notice("フォルダが選択されなかったため終了します。");
    return;
}

// 2. 移動先フォルダの選択（ここではルート階層 "/" も含める）
const allDestFolders = app.vault.getAllLoadedFiles().filter(f => f.children);
const destPaths = allDestFolders.map(f => f.path);
const destinationFolder = await tp.system.suggester(destPaths, allDestFolders, false, "移動先のフォルダを選択してください");

if (!destinationFolder) {
    new Notice("移動先が選択されなかったためキャンセルしました。");
    return;
}

// 3. エラー防止の除外処理
const originalCount = foldersToMove.length;
foldersToMove = foldersToMove.filter(f => {
    // ① 移動先フォルダ自身が対象リストに入っている場合は除外
    if (f.path === destinationFolder.path) return false;
    
    // ② 移動先が、移動対象フォルダの「中（サブフォルダ）」にある場合は除外（無限ループ/パスエラー防止）
    if (destinationFolder.path.startsWith(f.path + "/")) return false;
    
    return true;
});

const excludedCount = originalCount - foldersToMove.length;
if (excludedCount > 0) {
    new Notice(`${excludedCount}個のフォルダを移動対象から除外しました（移動先自身、またはエラー原因になるため）。`);
}

if (foldersToMove.length === 0) {
    new Notice("移動可能なフォルダがありませんでした。処理を終了します。");
    return;
}

// 4. 移動処理
let moveCount = 0;
for (let folder of foldersToMove) {
    // ルートが指定された場合のパス補正
    const destPathStr = destinationFolder.path === "/" ? "" : destinationFolder.path;
    const newPath = `${destPathStr}/${folder.name}`;
    
    // 既に同じ場所にいる場合はスキップ
    if (folder.path === newPath) continue;

    try {
        await app.fileManager.renameFile(folder, newPath);
        moveCount++;
    } catch (e) {
        new Notice(`エラー: 「${folder.name}」の移動に失敗しました。`);
        console.error(e);
    }
}

const destName = destinationFolder.path === "/" ? "ルート" : destinationFolder.name;
new Notice(`完了: ${moveCount} 個のフォルダを「${destName}」へ移動しました。`);
