<%*
// 1. 既存の全フォルダを取得してソート
const folders = app.vault.getAllLoadedFiles()
    .filter(f => f.children)
    .map(f => f.path)
    .sort();

// 親フォルダを最初に1回だけ選択
const parentFolder = await tp.system.suggester(folders, folders, false, "親フォルダ（作成先）を選択してください");
if (!parentFolder) return;

// 2. 連続してフォルダを作成するための無限ループ
while (true) {
    // フォルダ名（兼ファイル名）の入力要求
    const newName = await tp.system.prompt("作成するフォルダ名を入力（空欄またはキャンセルで終了）:");
    
    // 入力が空、またはEsc/キャンセルされた場合はループを抜けて終了
    if (!newName || newName.trim() === "") {
        new Notice("連続作成モードを終了しました。");
        break;
    }

    // パスの構築
    const targetFolderPath = parentFolder === "/" ? newName : `${parentFolder}/${newName}`;
    const targetFilePath = `${targetFolderPath}/${newName}.md`;

    // 3. フォルダの自動生成
    if (!await app.vault.adapter.exists(targetFolderPath)) {
        await app.vault.createFolder(targetFolderPath);
    }

    // 4. ファイルの自動生成と Waypoint の安全な書き込み
    if (!await app.vault.adapter.exists(targetFilePath)) {
        const fileContent = "%%" + " Waypoint " + "%%"; 
        const newFile = await app.vault.create(targetFilePath, fileContent);
        
        // 【改善】'tab' を指定することで、現在の画面を維持しつつ新しいタブで開く
        await app.workspace.getLeaf('tab').openFile(newFile);
        
        new Notice(`作成成功: ${newName}`);
    } else {
        new Notice(`スキップ: 「${newName}」は既に存在します。`);
    }
}
-%>
